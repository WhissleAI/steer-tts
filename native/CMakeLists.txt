cmake_minimum_required(VERSION 3.16)
project(steertts_edge_native VERSION 1.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# ONNX Runtime: a prebuilt release (https://github.com/microsoft/onnxruntime/releases, onnxruntime-<os>-<arch>-<ver>.tgz)
# unpacked anywhere; pass -DONNXRUNTIME_ROOT=<dir>. Without it, the matching release is downloaded at configure time.
set(ONNXRUNTIME_VERSION "1.31.0" CACHE STRING "ONNX Runtime release to download when ONNXRUNTIME_ROOT is not set")
set(ONNXRUNTIME_ROOT "" CACHE PATH "Unpacked ONNX Runtime release (include/, lib/)")
if(NOT ONNXRUNTIME_ROOT)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
    set(_ort_arch aarch64)
  else()
    set(_ort_arch x64)
  endif()
  set(_ort_name onnxruntime-linux-${_ort_arch}-${ONNXRUNTIME_VERSION})
  set(ONNXRUNTIME_ROOT ${CMAKE_BINARY_DIR}/${_ort_name})
  if(NOT EXISTS ${ONNXRUNTIME_ROOT}/include)
    message(STATUS "Downloading ${_ort_name}")
    file(DOWNLOAD https://github.com/microsoft/onnxruntime/releases/download/v${ONNXRUNTIME_VERSION}/${_ort_name}.tgz
         ${CMAKE_BINARY_DIR}/${_ort_name}.tgz SHOW_PROGRESS STATUS _st)
    list(GET _st 0 _code)
    if(NOT _code EQUAL 0)
      message(FATAL_ERROR "ONNX Runtime download failed (${_st}); pass -DONNXRUNTIME_ROOT=<unpacked release>")
    endif()
    file(ARCHIVE_EXTRACT INPUT ${CMAKE_BINARY_DIR}/${_ort_name}.tgz DESTINATION ${CMAKE_BINARY_DIR})
  endif()
endif()
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS ${ONNXRUNTIME_ROOT}/include ${ONNXRUNTIME_ROOT}/include/onnxruntime NO_DEFAULT_PATH)
find_library(ORT_LIB onnxruntime PATHS ${ONNXRUNTIME_ROOT}/lib NO_DEFAULT_PATH)
if(NOT ORT_INCLUDE OR NOT ORT_LIB)
  message(FATAL_ERROR "ONNX Runtime not found under ${ONNXRUNTIME_ROOT}")
endif()

add_subdirectory(src/front)

add_library(steertts STATIC src/engine.cpp src/dsp.cpp src/c_api.cpp)
target_include_directories(steertts PUBLIC include PRIVATE src ${ORT_INCLUDE})
target_link_libraries(steertts PUBLIC steertts_front ${ORT_LIB})

add_executable(steertts-edge src/cli.cpp)
target_link_libraries(steertts-edge PRIVATE steertts)
target_include_directories(steertts-edge PRIVATE ${ORT_INCLUDE})
# find libonnxruntime.so next to the binary (install / tarball layout) or in the build's ONNX Runtime dir
set_target_properties(steertts-edge PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib" INSTALL_RPATH "$ORIGIN/../lib")

add_executable(c_example examples/c_example.c)
set_target_properties(c_example PROPERTIES LINKER_LANGUAGE CXX BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
target_link_libraries(c_example PRIVATE steertts)
enable_language(C)

option(STEERTTS_TESTS "Build the tests" ON)
if(STEERTTS_TESTS)
  enable_testing()
  add_executable(engine_parity tests/engine_parity.cpp)
  target_link_libraries(engine_parity PRIVATE steertts)
  set_target_properties(engine_parity PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/front/CMakeLists.txt)
    add_subdirectory(tests/front)
  endif()
  # ctest needs a model: cmake -B build -DSTEERTTS_TEST_MODEL=/path/to/steer-tts-edge
  set(STEERTTS_TEST_MODEL "" CACHE PATH "steer-tts-edge package directory for ctest")
  if(STEERTTS_TEST_MODEL)
    add_test(NAME front_golden COMMAND front_golden ${STEERTTS_TEST_MODEL}/native/front
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/front/golden.jsonl ${CMAKE_CURRENT_SOURCE_DIR}/tests/front/golden_g2p.jsonl)
    add_test(NAME engine_parity COMMAND engine_parity ${STEERTTS_TEST_MODEL} ${CMAKE_CURRENT_SOURCE_DIR}/tests/engine_golden)
    add_test(NAME c_example COMMAND c_example ${STEERTTS_TEST_MODEL} "Hello. [pause 200ms] <whisper>Quietly.</whisper>"
             ${CMAKE_BINARY_DIR}/c_example.wav)
  endif()
endif()

install(TARGETS steertts-edge RUNTIME DESTINATION bin)
install(FILES ${ORT_LIB} DESTINATION lib)
file(GLOB _ort_sos ${ONNXRUNTIME_ROOT}/lib/libonnxruntime.so*)
install(FILES ${_ort_sos} DESTINATION lib)
install(FILES include/steertts/steertts.h include/steertts/steertts_c.h include/steertts/front.h DESTINATION include/steertts)
install(TARGETS steertts steertts_front ARCHIVE DESTINATION lib)
